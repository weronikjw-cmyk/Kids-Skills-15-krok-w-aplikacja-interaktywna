    <script>
        const apiKey = ""; 
        let currentStepId = '';
        let currentStepName = '';

        const inputs = document.querySelectorAll('.step-input');
        const actionButtons = document.getElementById('action-buttons');
        const hint = document.getElementById('progress-hint');

        function checkCompletion() {
            let filledCount = 0;
            inputs.forEach(input => {
                if (input.value.trim().length > 0) filledCount++;
            });

            if (filledCount >= 15) {
                actionButtons.classList.remove('hidden');
                hint.innerText = "Brawo! Możesz teraz wygenerować ściągawkę!";
            } else {
                actionButtons.classList.add('hidden');
                hint.innerText = `Uzupełniono: ${filledCount}/15. Wypełnij wszystko, by odblokować dodatki!`;
            }
        }

        function generateSummary() {
            const container = document.getElementById('summary-table-container');
            const summarySection = document.getElementById('summary-section');
            
            let tableHtml = `
                <table class="w-full text-left border-collapse">
                    <thead>
                        <tr class="bg-indigo-600 text-white">
                            <th class="p-4 text-xs uppercase tracking-wider border-b border-indigo-700">Krok</th>
                            <th class="p-4 text-xs uppercase tracking-wider border-b border-indigo-700">Zadanie</th>
                            <th class="p-4 text-xs uppercase tracking-wider border-b border-indigo-700">Moja Odpowiedź</th>
                        </tr>
                    </thead>
                    <tbody class="divide-y divide-slate-100">
            `;

            inputs.forEach(input => {
                const stepNum = input.getAttribute('data-step');
                const label = input.getAttribute('data-label');
                const val = input.value || "---";
                
                tableHtml += `
                    <tr class="hover:bg-slate-50 transition-colors">
                        <td class="p-4 font-bold text-indigo-500 w-16 text-center">${stepNum}</td>
                        <td class="p-4 font-semibold text-slate-700 text-sm w-48">${label}</td>
                        <td class="p-4 text-slate-600 text-sm italic">${val}</td>
                    </tr>
                `;
            });

            tableHtml += `</tbody></table>`;
            container.innerHTML = tableHtml;
            summarySection.style.display = 'block';
            
            // Przewiń do ściągawki
            summarySection.scrollIntoView({ behavior: 'smooth' });
        }

        inputs.forEach(el => {
            el.addEventListener('input', function() {
                this.style.height = 'auto';
                this.style.height = (this.scrollHeight) + 'px';
                checkCompletion();
            });
        });

        // --- FAJERWERKI ---
        const canvas = document.getElementById('fireworks-canvas');
        const ctx = canvas.getContext('2d');
        let particles = [];
        let animationFrame;

        function resizeCanvas() {
            canvas.width = window.innerWidth;
            canvas.height = window.innerHeight;
        }
        window.addEventListener('resize', resizeCanvas);
        resizeCanvas();

        class Particle {
            constructor(x, y, color) {
                this.x = x;
                this.y = y;
                this.color = color;
                this.velocity = {
                    x: (Math.random() - 0.5) * 8,
                    y: (Math.random() - 0.5) * 8
                };
                this.alpha = 1;
                this.friction = 0.95;
            }
            draw() {
                ctx.globalAlpha = this.alpha;
                ctx.beginPath();
                ctx.arc(this.x, this.y, 3, 0, Math.PI * 2);
                ctx.fillStyle = this.color;
                ctx.fill();
            }
            update() {
                this.velocity.x *= this.friction;
                this.velocity.y *= this.friction;
                this.velocity.y += 0.05; 
                this.x += this.velocity.x;
                this.y += this.velocity.y;
                this.alpha -= 0.01;
            }
        }

        function createFirework(x, y) {
            const colors = ['#ff0000', '#00ff00', '#0000ff', '#ffff00', '#ff00ff', '#00ffff', '#ffffff', '#ffaa00'];
            const color = colors[Math.floor(Math.random() * colors.length)];
            for (let i = 0; i < 40; i++) {
                particles.push(new Particle(x, y, color));
            }
        }

        function animateFireworks() {
            if (particles.length === 0) {
                ctx.clearRect(0, 0, canvas.width, canvas.height);
                return;
            }
            animationFrame = requestAnimationFrame(animateFireworks);
            ctx.clearRect(0, 0, canvas.width, canvas.height);
            
            particles.forEach((p, i) => {
                if (p.alpha > 0) {
                    p.update();
                    p.draw();
                } else {
                    particles.splice(i, 1);
                }
            });
        }

        function triggerCelebration() {
            animateFireworks();
            let count = 0;
            const interval = setInterval(() => {
                createFirework(Math.random() * canvas.width, Math.random() * (canvas.height * 0.6));
                count++;
                if (count > 25) {
                    clearInterval(interval);
                    setTimeout(() => {
                        document.getElementById('congrats-overlay').style.display = 'flex';
                    }, 1000);
                }
            }, 250);
        }

        function closeModal() {
            document.getElementById('gemini-modal').classList.add('hidden');
        }

        async function callGemini(prompt, system) {
            const url = `https://generativelanguage.googleapis.com/v1beta/models/gemini-2.5-flash-preview-09-2025:generateContent?key=${apiKey}`;
            const payload = {
                contents: [{ parts: [{ text: prompt }] }],
                systemInstruction: { parts: [{ text: system }] }
            };

            try {
                const response = await fetch(url, {
                    method: 'POST',
                    headers: { 'Content-Type': 'application/json' },
                    body: JSON.stringify(payload)
                });
                const data = await response.json();
                return data.candidates?.[0]?.content?.parts?.[0]?.text || "Przepraszam, nie mogłem teraz wygenerować sugestii. Spróbuj jeszcze raz.";
            } catch (err) {
                return "Błąd połączenia. Sprawdź internet.";
            }
        }
